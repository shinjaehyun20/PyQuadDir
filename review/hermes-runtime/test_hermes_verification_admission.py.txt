class FirstGoalBootstrapTests(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        import importlib.util
        source = v.ADAPTER
        spec = importlib.util.spec_from_file_location('bootstrap_adapter', source)
        cls.adapter = importlib.util.module_from_spec(spec)
        # Execute exactly the source bytes included in the verification binding,
        # never a timestamp-valid .pyc with a different source identity.
        exec(compile(source.read_bytes(), str(source), 'exec'), cls.adapter.__dict__)

    def setUp(self):
        self.argv = [str(v.APPROVED_EXECUTABLE), str(Path(g.__file__).resolve()), '--runtime', 'hermes',
                     '--session-id', 'fixture-first-session', '--action', 'lock-goal',
                     '--prompt', 'Repair first goal metadata bootstrap with bounded native pytest regression tests']
        self.payload = {'session_id': 'fixture-first-session', 'tool_name': 'terminal',
                        'cwd': 'D:/workspace', 'tool_input': {'workdir': 'D:/workspace',
                        'command': self.command()}}
        self.registry = {'tasks': {}, 'active_by_session': {}}

    def command(self):
        return ' '.join(shlex.quote(s.replace('\\', '/')) for s in self.argv)

    def run_hook(self, task=None):
        output = io.StringIO()
        a = self.adapter
        with patch.object(sys, 'argv', ['adapter', '--runtime', 'hermes', '--event', 'NativePreToolUse']), \
             patch.object(sys, 'stdin', io.StringIO(json.dumps(self.payload)+'\n')), \
             patch.object(a.native, 'load_registry', return_value=self.registry), \
             patch.object(a.native, 'active_task', return_value=task), \
             patch.object(a.native, 'is_mutating_tool', return_value=True), \
             patch.object(a, 'scoped_cache', return_value={'outcome':'STALE', 'allow_generation':False, 'reason':'fixture'}), \
             contextlib.redirect_stdout(output):
            self.assertTrue(a.native.is_mutating_tool('hermes', 'terminal', self.payload['tool_input']))
            self.assertEqual(a.main(), 0)
        return output.getvalue()

    def test_first_same_session_native_lock_reaches_mutating_branch(self):
        self.assertTrue(self.adapter.first_goal_bootstrap(self.payload, self.registry, 'fixture-first-session'))
        self.assertEqual(self.run_hook(), '')

    def test_canonical_caller_remains_allowed_with_different_hook_host(self):
        with patch.object(sys, 'executable', 'C:/different-hook-host/python.exe'):
            self.assertEqual(self.run_hook(), '')
            self.argv[0] = sys.executable
            self.payload['tool_input']['command'] = self.command()
            self.assertEqual(json.loads(self.run_hook())['action'], 'block')

    def test_wrong_session_runtime_script_executable_extra_arguments_and_chain_denied(self):
        original = copy.deepcopy(self.argv)
        for index, value in [(0,'not-the-native-python'), (1,'D:/workspace/tools/stack/fake.py'),
                             (3,'codex'), (5,'other-session'), (7,'update-task'), (9,'')]:

                self.argv = copy.deepcopy(original); self.argv[index] = value
                self.payload['tool_input']['command'] = self.command()
                self.assertEqual(json.loads(self.run_hook())['action'], 'block')
        self.argv = original
        for suffix in [' --task-id other', ' --runtime hermes', '; whoami', ' && whoami',
                       ' | whoami', ' > D:/workspace/output/forbidden', '\nwhoami']:
            with self.subTest(suffix=suffix):
                self.payload['tool_input']['command'] = self.command()+suffix
                self.assertEqual(json.loads(self.run_hook())['action'], 'block')

    def test_existing_owner_pointer_and_finished_claim_denied(self):
        for status in ['active','done']:
            self.registry['tasks'] = {'existing': {'runtime':'hermes', 'owner_session_id':'fixture-first-session', 'status':status}}

class CompositionTests(unittest.TestCase):
    """Actual projection, consumer, mutation classifier and synthetic cache rows.

    Only registry locations and stdin are substituted, never cache decisions.
    """
    @classmethod
    def setUpClass(cls):
        FirstGoalBootstrapTests.setUpClass()
        cls.adapter = FirstGoalBootstrapTests.adapter

    def setUp(self):
        import tempfile
        AdmissionTests.setUp(self)
        base = getattr(v, 'OFFLINE_SCOPE', v.WORKSPACE / 'tmp' / 'hermes-admission-composition-tests')
        if not hasattr(v, 'OFFLINE_SCOPE'):base.mkdir(exist_ok=True)
        self.root = Path(tempfile.mkdtemp(prefix='composition-', dir=base))
        self.source = self.root / 'source.txt'
        with self.source.open('x') as stream:stream.write('actual synthetic pointer bytes')
        self.cache_path = self.root / 'golden-fixture.json'
        self.row = {'case_id':'composition-case', 'kind':'artifact',
            'canonical_path':str(self.source), 'sha256':'0'*64,
            'input_fingerprint':v.digest(self.source), 'verifier_ref':'synthetic-only',
            'summary':'actual adapter composition regression', 'graph_refs':[]}
        self.task['cache_policy'].update(query='actual adapter composition regression',
            case_id=self.row['case_id'], input_fingerprint=self.row['input_fingerprint'],
            inputs=[{'path':str(self.source),'sha256':v.digest(self.source)}])
        self.registry = {'tasks':{self.task['task_id']:self.task},
            'active_by_session':{'hermes':{'fixture-session':self.task['task_id']}}}

    def write_cache(self, data=None):
        with self.cache_path.open('x') as stream:
            json.dump(data if data is not None else {'schema':self.adapter.cache.SCHEMA,'cases':[self.row]}, stream)

    def run_actual_hook(self, payload=None):
        payload = self.payload if payload is None else payload
        output = io.StringIO()
        a = self.adapter
        # main installs a global cache consumer; restore it after each invocation.
        with patch.object(sys,'argv',['adapter','--runtime','hermes','--event','NativePreToolUse']), \
             patch.object(sys,'stdin',io.StringIO(json.dumps(payload)+'\n')), \
             patch.object(a.native,'load_registry',return_value=self.registry), \
             patch.object(a.native,'WARP_CACHE_REGISTRY',self.cache_path), \
             patch.object(a.native,'cache_gate_result',a.native.cache_gate_result), \
             contextlib.redirect_stdout(output):
            self.assertEqual(a.main(),0)
        return output.getvalue()

    def test_actual_stale_is_advisory_for_verifier_and_ordinary_mutation(self):
        self.write_cache()
        before = v.digest(self.cache_path)
        raw = self.adapter.cache.evaluate(self.cache_path,'composition',case_id=self.row['case_id'])
        self.assertEqual(raw['outcome'],'STALE')
        projected = self.adapter.projected(self.task,self.payload,self.cache_path)
        self.assertEqual(projected['outcome'],'MISS')
        self.assertTrue(projected['allow_generation'])
        self.assertFalse(projected['task_complete'])
        with patch.object(v,'admits_verification',wraps=v.admits_verification) as admission:
            self.assertEqual(self.run_actual_hook(),'')
            for name in ['write_file','patch','terminal']:
                ordinary = {'session_id':'fixture-session','tool_name':name,
                    'tool_input':{'path':str(self.root/'not-created.txt'),
                                  'command':'python -c "print(123)"'}}
                self.assertEqual(self.run_actual_hook(ordinary),'')
            admission.assert_not_called()
        self.assertFalse((self.root/'not-created.txt').exists())
        self.assertEqual(v.digest(self.cache_path),before)
        self.assertEqual(self.adapter.cache.evaluate(self.cache_path,'composition',case_id=self.row['case_id'])['outcome'],'STALE')

    def test_actual_projection_preserves_error_and_input_denials(self):
        for outcome in ['ERROR','INPUT_REQUIRED','INPUT_MISMATCH','HIT_ARTIFACT','NOT_FOUND']:
            with self.subTest(outcome=outcome):
                self.cache_path = self.root / (outcome+'.json')
                self.row['sha256'] = v.digest(self.source)
                policy = self.task['cache_policy']
                policy['input_fingerprint'] = self.row['input_fingerprint']
                policy['case_id'] = self.row['case_id']
                if outcome == 'INPUT_REQUIRED':policy['input_fingerprint']=''
                if outcome == 'INPUT_MISMATCH':policy['input_fingerprint']='different-synthetic-input'
                if outcome == 'NOT_FOUND':policy['case_id']='absent-case'
                self.write_cache({'schema':'invalid','cases':[]} if outcome == 'ERROR' else None)
                self.assertEqual(self.adapter.projected(self.task,self.payload,self.cache_path)['outcome'],outcome)
                for payload in [self.payload, {'session_id':'fixture-session','tool_name':'write_file',
                     'tool_input':{'path':str(self.root/'never-created')}}]:
                    self.assertIn('Warp Cache '+outcome,json.loads(self.run_actual_hook(payload))['message'])

    def test_actual_projection_source_mismatch_and_owner_mode_blocks(self):
        self.write_cache()
        self.task['cache_policy']['inputs'][0]['sha256']='0'*64
        self.assertEqual(self.adapter.projected(self.task,self.payload,self.cache_path)['outcome'],'INPUT_MISMATCH')
        self.assertIn('INPUT_MISMATCH',json.loads(self.run_actual_hook())['message'])
        self.task['cache_policy']['inputs'][0]['sha256']=v.digest(self.source)
        for key,value in [('owner_session_id','other'),('runtime','codex'),('mode','plan'),
                          ('status','blocked'),('work_state','BLOCKED')]:
            with self.subTest(key=key), patch.dict(self.task,{key:value}):
                self.assertEqual(json.loads(self.run_actual_hook())['action'],'block')
        wrong = copy.deepcopy(self.payload);wrong['session_id']='other'
        self.assertEqual(json.loads(self.run_actual_hook(wrong))['action'],'block')

    def test_native_payload_shapes_use_same_normalizers_as_consumer(self):

        missing = copy.deepcopy(self.payload);del missing['session_id']
        self.assertFalse(v.admits_verification('hermes',missing,self.task))
        for payload in [self.payload,
                {'sessionId':'fixture-session','toolCall':{'name':'terminal','args':tool}},
                {'conversationId':'fixture-session','toolName':'terminal','toolInput':tool},
                {'session_id':'fixture-session','tool':'terminal','input':tool}]:
            with self.subTest(payload=payload):
                self.assertTrue(v.admits_verification('hermes',payload,self.task))
                output=io.StringIO()
                with patch.object(g,'active_task',return_value=self.task), \
                     patch.object(g,'cache_gate_result',return_value={'outcome':'STALE','allow_generation':False,'reason':'raw fixture'}), \
                     contextlib.redirect_stdout(output):
                    g.handle_pre_tool('hermes',payload,registry={})
                self.assertEqual(output.getvalue(),'')
                changed=copy.deepcopy(payload)
                g.get_tool_input(changed)['workdir']=str(v.ROOT)
                self.assertFalse(v.admits_verification('hermes',changed,self.task))

    def test_all_metadata_aliases_resolve_to_canonical_cli(self):
        import shutil
        policy={'query':'goal_registry.py','input_fingerprint':'synthetic-fixture',
                'inputs':self.binding['inputs']}
        def payload(exe):
            argv=[exe,str(v.ROOT/'goal_registry.py'),'--runtime','hermes','--action','update-task',
                  '--session-id','fixture-session','--task-id',self.task['task_id'],
                  '--set-json',json.dumps({'cache_policy':policy})]
            return {'session_id':'fixture-session','tool_name':'terminal',
                    'tool_input':{'command':' '.join(shlex.quote(s) for s in argv)}}
        for exe in ['python','python.exe','python3','python3.exe','py','py.exe',str(v.APPROVED_EXECUTABLE)]:
            for resolved in [str(v.APPROVED_EXECUTABLE),'C:/wrong/python.exe',None]:
                with self.subTest(exe=exe,resolved=resolved), patch.object(shutil,'which',return_value=resolved):
                    self.assertEqual(g.is_same_session_cache_policy_registration('hermes',payload(exe),self.task),
                                     resolved==str(v.APPROVED_EXECUTABLE))
        # Real PATH evidence, not an assumption about the shell's Python alias.
        for exe in ['python','python.exe','py',str(v.APPROVED_EXECUTABLE)]:
            resolved=shutil.which(exe)
            expected=bool(resolved) and Path(resolved).resolve()==v.APPROVED_EXECUTABLE.resolve()
            self.assertEqual(g.is_same_session_cache_policy_registration('hermes',payload(exe),self.task),expected)
