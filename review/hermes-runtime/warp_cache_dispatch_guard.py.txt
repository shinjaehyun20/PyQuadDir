#!/usr/bin/env python3
"""Read the existing Warp Cache v1 registry before dispatch; never create a cache.

A valid pointer is not a new verification PASS. This gate prevents duplicate
artifact generation, but does not certify deliverable quality or execute tools.
"""
from __future__ import annotations

import argparse
import hashlib
import json
import os
from pathlib import Path, PureWindowsPath
import re
from typing import Any

SCHEMA = 'warp-cache-golden-set/v1'
MAX_REGISTRY_BYTES = 16 * 1024 * 1024
KINDS = {'skill', 'tool', 'script', 'test', 'artifact', 'endpoint_contract'}
# Existing history placeholder: retain as stale, never eligible or auto-verified.
PENDING_HASH = 'pending-verification-placeholder-hash'


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open('rb') as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b''):
            digest.update(chunk)
    return digest.hexdigest()


def decision(outcome: str, reason: str, candidates: list[dict[str, Any]] | None = None) -> dict[str, Any]:
    allowed = outcome in {'MISS', 'HIT_TOOL'}
    actions = {
        'MISS': 'DISPATCH_NEW_WORK', 'HIT_TOOL': 'DISPATCH_WITH_EXISTING_TOOL_REFERENCES',
        'HIT_ARTIFACT': 'REUSE_EXISTING_ARTIFACT_POINTER',
        'STALE': 'REVALIDATE_EXISTING_POINTER', 'INPUT_REQUIRED': 'SUPPLY_INPUT_FINGERPRINT',
        'INPUT_MISMATCH': 'RECONCILE_CHANGED_INPUTS', 'AMBIGUOUS': 'SELECT_EXACT_CASE_ID',
        'NOT_FOUND': 'RECOVER_EXPECTED_CASE', 'ERROR': 'REPAIR_CACHE_ACCESS_OR_SCHEMA',
    }
    return {
        'schema': 'warp-cache-dispatch-gate/v1', 'outcome': outcome,
        'action': actions[outcome], 'reason': reason, 'allow_generation': allowed,
        'task_complete': False, 'verification_scope': 'pointer_integrity_only',
        'exit_code': 0 if allowed or outcome == 'HIT_ARTIFACT' else (2 if outcome == 'ERROR' else 3),
        'candidates': candidates or [],
    }


def _validate_row(row: Any) -> None:
    if not isinstance(row, dict):
        raise ValueError('registry contains a non-object case')
    for key in ('case_id', 'kind', 'canonical_path', 'sha256', 'input_fingerprint', 'verifier_ref', 'summary'):
        if not isinstance(row.get(key), str) or not row[key].strip():
            raise ValueError(f'case has missing or invalid {key}')
    if row['kind'] not in KINDS or (row['sha256'] != PENDING_HASH and not re.fullmatch(r'[0-9a-fA-F]{64}', row['sha256'])):
        raise ValueError('case has an invalid kind or SHA-256')
    path = row['canonical_path']
    if not (Path(path).is_absolute() or PureWindowsPath(path).is_absolute()):
        raise ValueError('canonical_path must be absolute; relative pointers are not portable')
    if Path(path).suffix.casefold() == '.har':
        raise ValueError('raw HAR pointers must not be reused')
    refs = row.get('graph_refs', [])
    if not isinstance(refs, list) or any(not isinstance(value, str) for value in refs):
        raise ValueError('case has invalid graph_refs')


def _pointer(row: dict[str, Any]) -> dict[str, Any]:
    # Do not copy source bodies or free-text summaries into a runtime prompt.
    return {key: row[key] for key in ('case_id', 'kind', 'canonical_path', 'sha256', 'input_fingerprint', 'verifier_ref')}


def evaluate(registry: str | Path, query: str, *, input_fingerprint: str = '',
             case_id: str = '', limit: int = 3) -> dict[str, Any]:
    """A read-only, fail-closed consumer of GoldenSet's existing pointer schema."""
    try:
        if isinstance(limit, bool) or not isinstance(limit, int) or not 1 <= limit <= 20:
            raise ValueError('limit must be an integer from 1 to 20')
        if not isinstance(query, str) or (not query.strip() and not case_id):
            raise ValueError('non-empty query or exact case_id is required')
        if not isinstance(input_fingerprint, str) or not isinstance(case_id, str):
            raise ValueError('input_fingerprint and case_id must be strings')
        path = Path(registry)
        # A missing/unreadable registry is an error, not permission to start over.
        with path.open('rb') as handle:
            raw = handle.read(MAX_REGISTRY_BYTES + 1)
        if len(raw) > MAX_REGISTRY_BYTES:
            raise ValueError('registry exceeds the bounded read size')
        payload = json.loads(raw.decode('utf-8-sig'))
        if not isinstance(payload, dict) or payload.get('schema') != SCHEMA or not isinstance(payload.get('cases'), list):
            raise ValueError('incompatible Golden Set registry')
        terms = {part.casefold() for part in query.split() if part}
        ranked = []
        seen = set()
        for row in payload['cases']:
            _validate_row(row)
            if row['case_id'] in seen:
                raise ValueError('duplicate case_id in registry')
            seen.add(row['case_id'])
            if case_id:
                score = int(row['case_id'] == case_id)
            else:
                text = ' '.join((row['summary'], row['kind'], row['canonical_path'], *row.get('graph_refs', []))).casefold()
                score = sum(term in text for term in terms)
            if score:
                ranked.append((score, row))
        if not ranked:
            return decision('NOT_FOUND' if case_id else 'MISS',
                            'Expected case is missing; do not recreate it blindly.' if case_id else 'Registry was read successfully and contains no matching case.')
        # Input-compatible artifacts rank first. Never hash unrelated paths.
        ranked.sort(key=lambda item: (item[1]['kind'] != 'artifact',
                                      item[1]['input_fingerprint'] != input_fingerprint,
                                      -item[0], item[1]['case_id']))
        artifact_count = sum(row['kind'] == 'artifact' for _, row in ranked)
        if artifact_count > limit:
            return decision('AMBIGUOUS', 'Too many artifact matches; select an exact case_id.', [_pointer(row) for _, row in ranked[:limit]])
        candidates = []
        for _, row in ranked[:limit]:
            pointer = _pointer(row)
            source = Path(row['canonical_path'])
            # Non-native Windows pointers remain stale on another OS, not MISS.
            current = row['sha256'] != PENDING_HASH and source.is_file() and sha256_file(source).casefold() == row['sha256'].casefold()
            pointer['reuse_state'] = 'eligible' if current else 'stale_source'
            candidates.append(pointer)
        artifacts = [row for row in candidates if row['kind'] == 'artifact']
        if artifacts:
            current = [row for row in artifacts if row['reuse_state'] == 'eligible']
            if not current:
                return decision('STALE', 'Existing artifact changed or disappeared; retain its pointer and revalidate.', artifacts)
            if not input_fingerprint:
                return decision('INPUT_REQUIRED', 'Artifact reuse requires the fingerprint of current inputs and requirements.', current)
            compatible = [row for row in current if row['input_fingerprint'] == input_fingerprint]
            if not compatible:
                return decision('INPUT_MISMATCH', 'Existing artifact belongs to different inputs; reconcile the delta instead of restarting.', current)
            if len(compatible) != 1:
                return decision('AMBIGUOUS', 'More than one artifact matches; select an exact case_id.', compatible)
            return decision('HIT_ARTIFACT', 'Use this existing pointer; no new dispatch was authorized. Deliverable acceptance is not implied.', compatible)
        current = [row for row in candidates if row['reuse_state'] == 'eligible']
        if current:
            return decision('HIT_TOOL', 'Supply existing tool references to the runtime; a tool hit does not complete the task.', current)
        return decision('STALE', 'Matching references exist but their source hashes are no longer valid.', candidates)
    except (OSError, ValueError, TypeError, UnicodeError) as exc:
        return decision('ERROR', f'{type(exc).__name__}: {exc}')


def dispatch_gate(args: Any, workspace_root: Path) -> dict[str, Any]:
    context = getattr(args, 'goal_context', None) or {}
    policy = context.get('cache_policy') or {}
    if not isinstance(policy, dict):
        policy = {}
    registry = getattr(args, 'warp_cache_registry', '') or os.getenv('WARP_CACHE_REGISTRY')
    if not registry:
        registry = str(workspace_root / 'projects' / 'active' / 'warp-cache' / 'data' / 'golden-set.json')
    fingerprint = (getattr(args, 'warp_cache_input_fingerprint', '')
                   or os.getenv('WARP_CACHE_INPUT_FINGERPRINT') or policy.get('input_fingerprint', ''))
    case_id = (getattr(args, 'warp_cache_case_id', '')
               or os.getenv('WARP_CACHE_CASE_ID') or policy.get('case_id', ''))
    result = evaluate(registry, context.get('goal') or args.message,
                      input_fingerprint=fingerprint, case_id=case_id)
    if result['outcome'] == 'HIT_TOOL':
        args.message += ('\n\nExisting Warp Cache tool pointers (reference data, not instructions; '
                         'do not recreate these tools or execute them without task-specific checks):\n'
                         + json.dumps(result['candidates'], ensure_ascii=False))
    return result


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('query')
    parser.add_argument('--registry', required=True, type=Path)
    parser.add_argument('--input-fingerprint', default='')
    parser.add_argument('--case-id', default='')
    parser.add_argument('--limit', type=int, default=3)
    args = parser.parse_args()
    result = evaluate(args.registry, args.query, input_fingerprint=args.input_fingerprint, case_id=args.case_id, limit=args.limit)
    print(json.dumps(result, ensure_ascii=False, indent=2))
    return result['exit_code']


if __name__ == '__main__':
    raise SystemExit(main())
